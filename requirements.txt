# --- PII filter (Phase 1) ---
# Pattern-recognizer based, no spaCy/NLP engine dependency: the spaCy
# tokenizer wheels available for Python 3.12 on Windows hit a binary
# compatibility bug (OverflowError in spacy.tokenizer.Tokenizer._reset_cache)
# on every model tested, including a blank pipeline. Presidio's
# PatternRecognizer + AnonymizerEngine run standalone without it.
#
# spacy/thinc are still installed below, not because we use them (we
# don't), but because presidio_analyzer unconditionally imports
# spacy.tokens at import time, it's a hard dependency of the package
# itself, not an optional NLP backend. Pinned to >=3.8 specifically
# because numpy 2.x (needed by onnx/fastembed in Phase 3) requires a
# thinc build with numpy-2 support, which only landed in spacy 3.8+.
presidio-analyzer==2.2.355
presidio-anonymizer==2.2.355
spacy>=3.8

# fastembed, not sentence-transformers: ONNX Runtime based, no torch
# dependency, avoiding the c10.dll native-library conflict that affected
# Phase 1/2. Keeps embeddings fully local/self-hosted too, consistent with
# this project's own compliance story rather than calling an external
# embedding API.
fastembed==0.4.2
# protobuf pinned >=6.31.1: onnx (via fastembed) requires it. mlflow was
# bumped to 2.22.5 specifically because mlflow 2.17's mlflow-skinny pinned
# protobuf<6, directly conflicting with onnx's floor. Both onnx and mlflow
# are genuinely used (embeddings, experiment tracking), this isn't a
# drop-the-unused-one fix like sentence-transformers/langchain were.
protobuf>=6.31.1
# langchain and langchain-community were removed: neither langgraph nor
# langchain-ollama actually requires them (both depend only on
# langchain-core, which has no numpy constraint at all), and the full
# langchain/langchain-community packages pin numpy<2.0, which directly
# conflicts with onnx's numpy>=2.0 floor. Same unused-speculative-dependency
# pattern as sentence-transformers, added in Phase 0 before any RAG code
# existed, never actually imported.
#
# LLM generation runs through a local Ollama instance (llama3.2, already
# pulled), not an external API, for the same data-locality reason as the
# embedding model choice. Ollama's default context window for this model
# attempts to allocate ~14GB of KV cache and fails outright on this
# machine; src/rag/agent.py explicitly bounds num_ctx to 4096.
